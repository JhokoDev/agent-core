def main() -> None:
    parser = argparse.ArgumentParser(description="PreparaÃ§Ã£o e diagnÃ³stico â€” Fase 0")
    parser.add_argument("command", choices=["serve", "doctor", "smoke-model"])
    args = parser.parse_args()
    try:
        settings = Settings()
    except ValidationError:
        print("ConfiguraÃ§Ã£o invÃ¡lida. Execute scripts/bootstrap.py e confira .env; valores omitidos por seguranÃ§a.", file=sys.stderr)
        raise SystemExit(2)
    if args.command == "serve":
        import uvicorn
        from app.main import create_app
        uvicorn.run(create_app(settings), host=settings.host, port=settings.port, access_log=False)
        return

    async def check() -> int:
        provider = create_provider(settings)
        try:
            health = await provider.health()
            if args.command == "doctor":
                db = Database(settings.database_path)
                db.initialize()
                db_ok = db.ready()
                print(json.dumps({"database": "ready" if db_ok else "unavailable", "llm": health.model_dump()}, ensure_ascii=False))
                return 0 if db_ok and health.status == "ready" else 1
            if health.status != "ready":
                print(json.dumps(health.model_dump()))
                return 1
            result = await provider.chat([Message(role="user", content="Responda em portuguÃªs, em uma frase: o que Ã© um computador?")])
            # ExibiÃ§Ã£o direta solicitada pelo operador; nÃ£o grava conteÃºdo em logs.
            print(result.content)
            return 0
        except ProviderError as exc:
            print(str(exc), file=sys.stderr)
            return 1
        finally:
            await provider.aclose()

    raise SystemExit(asyncio.run(check()))

def preparar_contexto_limpo(
    mensagens: list[dict],
    limite: int = 20,
) -> list[dict]:
    """
    MantÃ©m apenas mensagens vÃ¡lidas e limita o tamanho
    do histÃ³rico enviado ao modelo.
    """

    contexto: list[dict] = []

    for mensagem in mensagens[-limite:]:
        if not isinstance(mensagem, dict):
            continue

        role = mensagem.get("role")
        content = mensagem.get("content")

        if role not in {
            "system",
            "user",
            "assistant",
            "tool",
        }:
            continue

        if not isinstance(content, str):
            continue

        if not content.strip():
            continue

        contexto.append(
            {
                "role": role,
                "content": content,
            }
        )

    return contexto

def serializar_plano(plano: Any) -> dict | None:
    """
    Converte um PlannerContract em um dicionÃ¡rio serializÃ¡vel.
    """

    if plano is None:
        return None

    if hasattr(plano, "model_dump"):
        return plano.model_dump()

    if hasattr(plano, "__dict__"):
        resultado = {}

        for chave, valor in plano.__dict__.items():
            if hasattr(valor, "__dict__"):
                resultado[chave] = valor.__dict__

            elif isinstance(valor, list):
                resultado[chave] = [
                    item.__dict__
                    if hasattr(item, "__dict__")
                    else item
                    for item in valor
                ]

            else:
                resultado[chave] = valor

        return resultado

    return None

def processar_mensagem_ia_stream(
    mensagem: str,
    chat_id: str | None = None,
    skill: dict | None = None,
    imagem_base64: str | None = None,
):
    """
    Processa uma mensagem utilizando:

    Router
    â†’ Model Registry
    â†’ Memory
    â†’ Planner
    â†’ Tool Registry
    â†’ Permission System
    â†’ Model Provider
    """

    chat = _obter_chat(
        chat_id,
    )

    chat_id = chat["id"]

    mensagens_chat = chat.get(
        "messages",
        [],
    )

    # --------------------------------------------------------
    # SKILL
    # --------------------------------------------------------

    skill_id = None
    skill_prompt = None
    ferramentas_permitidas: list[str] = []

    if skill:
        skill_id = skill.get(
            "id",
        )

        skill_prompt = skill.get(
            "prompt",
            skill.get("system_prompt"),
        )

        ferramentas_permitidas = skill.get(
            "ferramentas_permitidas",
            skill.get("tools", []),
        ) or []

    # --------------------------------------------------------
    # CONTEXTO DE PERMISSÃ•ES
    # --------------------------------------------------------

    contexto_id = f"chat:{chat_id}"

    carregar_permissoes_skill(
        contexto_id,
        ferramentas_permitidas,
    )

    # --------------------------------------------------------
    # MENSAGEM DO USUÃRIO
    # --------------------------------------------------------

    conteudo_usuario = mensagem

    if imagem_base64:
        conteudo_usuario = (
            f"{mensagem}\n\n"
            "[Imagem anexada em Base64]"
        )

    mensagem_usuario = {
        "role": "user",
        "content": conteudo_usuario,
    }

    mensagens_chat.append(
        mensagem_usuario,
    )

    # --------------------------------------------------------
    # ROUTER
    # --------------------------------------------------------

    rota = ROUTER.rotear(
        mensagem,
    )

    modelo = rota.modelo

    # --------------------------------------------------------
    # MEMORY
    # --------------------------------------------------------

    memorias = recuperar_memorias(
        mensagem,
        limite=5,
    )

    memoria_contexto = ""

    if memorias:
        linhas_memoria = []

        for memoria in memorias:
            if isinstance(memoria, dict):
                conteudo = memoria.get(
                    "conteudo",
                    memoria.get("content", ""),
                )
            else:
                conteudo = getattr(
                    memoria,
                    "conteudo",
                    getattr(
                        memoria,
                        "content",
                        "",
                    ),
                )

            if conteudo:
                linhas_memoria.append(
                    f"- {conteudo}"
                )

        if linhas_memoria:
            memoria_contexto = (
                "MemÃ³rias relevantes:\n"
                + "\n".join(linhas_memoria)
            )

    # --------------------------------------------------------
    # PLANNER
    # --------------------------------------------------------

    plano = None

    if rota.rota == "complexo":
        try:
            plano = PLANNER.criar_plano(
                mensagem,
            )
        except Exception:
            plano = None

    # --------------------------------------------------------
    # SYSTEM PROMPT
    # --------------------------------------------------------

    mensagens_modelo: list[dict] = []

    system_parts = [
        "VocÃª Ã© o Agent Core, um assistente pessoal generalista.",
        "Responda de forma objetiva, precisa e Ãºtil.",
    ]

    if skill_prompt:
        system_parts.append(
            f"\nInstruÃ§Ãµes da Skill:\n{skill_prompt}"
        )

    if memoria_contexto:
        system_parts.append(
            f"\n{memoria_contexto}"
        )

    if plano:
        plano_serializado = serializar_plano(
            plano,
        )

        if plano_serializado:
            system_parts.append(
                "\nPlano de execuÃ§Ã£o:\n"
                + json.dumps(
                    plano_serializado,
                    ensure_ascii=False,
                )
            )

    mensagens_modelo.append(
        {
            "role": "system",
            "content": "\n".join(system_parts),
        }
    )

    # --------------------------------------------------------
    # HISTÃ“RICO
    # --------------------------------------------------------

    mensagens_modelo.extend(
        preparar_contexto_limpo(
            mensagens_chat,
        )
    )

    # --------------------------------------------------------
    # TOOLS
    # --------------------------------------------------------

    ferramentas = obter_ferramentas_da_skill(
        ferramentas_permitidas,
    )

    # --------------------------------------------------------
    # PRIMEIRA CHAMADA AO MODELO
    # --------------------------------------------------------

    resposta_modelo = MODEL_PROVIDER.gerar(
        modelo=modelo,
        mensagens=mensagens_modelo,
        temperatura=0.5,
        ferramentas=ferramentas or None,
    )

    # --------------------------------------------------------
    # PROCESSAMENTO DE TOOL CALLS
    # --------------------------------------------------------

    tool_calls = []

    mensagem_resposta = resposta_modelo.get(
        "message",
        {},
    )

    if isinstance(mensagem_resposta, dict):
        tool_calls = mensagem_resposta.get(
            "tool_calls",
            [],
        ) or []

    resultados_tools = []

    for tool_call in tool_calls:
        if not isinstance(tool_call, dict):
            continue

        function_data = tool_call.get(
            "function",
            {},
        )

        if not isinstance(function_data, dict):
            continue

        ferramenta_id = function_data.get(
            "name",
        )

        argumentos = function_data.get(
            "arguments",
            {},
        )

        if isinstance(argumentos, str):
            try:
                argumentos = json.loads(
                    argumentos,
                )
            except json.JSONDecodeError:
                argumentos = {}

        if not isinstance(argumentos, dict):
            argumentos = {}

        if not ferramenta_id:
            continue

        try:
            resultado = executar_ferramenta_autorizada(
                contexto_id=contexto_id,
                ferramenta_id=ferramenta_id,
                argumentos=argumentos,
            )

            resultados_tools.append(
                {
                    "tool_call": tool_call,
                    "resultado": resultado,
                    "erro": None,
                }
            )

        except Exception as erro:
            resultados_tools.append(
                {
                    "tool_call": tool_call,
                    "resultado": None,
                    "erro": str(erro),
                }
            )

    # --------------------------------------------------------
    # SEGUNDA CHAMADA AO MODELO
    # --------------------------------------------------------

    resposta_final = resposta_modelo

    if resultados_tools:
        mensagem_assistente = {
            "role": "assistant",
            "content": mensagem_resposta.get(
                "content",
                "",
            ),
            "tool_calls": tool_calls,
        }

        mensagens_modelo.append(
            mensagem_assistente,
        )

        for resultado_tool in resultados_tools:
            tool_call = resultado_tool[
                "tool_call"
            ]

            function_data = tool_call.get(
                "function",
                {},
            )

            ferramenta_id = function_data.get(
                "name",
                "",
            )

            resultado = resultado_tool[
                "resultado"
            ]

            erro = resultado_tool[
                "erro"
            ]

            if erro:
                conteudo_tool = json.dumps(
                    {
                        "erro": erro,
                    },
                    ensure_ascii=False,
                )

            else:
                try:
                    conteudo_tool = json.dumps(
                        resultado,
                        ensure_ascii=False,
                    )
                except TypeError:
                    conteudo_tool = str(
                        resultado,
                    )

            mensagens_modelo.append(
                {
                    "role": "tool",
                    "content": conteudo_tool,
                    "name": ferramenta_id,
                }
            )

        resposta_final = MODEL_PROVIDER.gerar(
            modelo=modelo,
            mensagens=mensagens_modelo,
            temperatura=0.5,
            ferramentas=ferramentas or None,
        )

    # --------------------------------------------------------
    # CONTEÃšDO FINAL
    # --------------------------------------------------------

    mensagem_final = resposta_final.get(
        "message",
        {},
    )

    if isinstance(mensagem_final, dict):
        conteudo_final = mensagem_final.get(
            "content",
            "",
        )
    else:
        conteudo_final = ""

    if not isinstance(
        conteudo_final,
        str,
    ):
        conteudo_final = str(
            conteudo_final,
        )

    # --------------------------------------------------------
    # PERSISTÃŠNCIA
    # --------------------------------------------------------

    mensagem_assistente_final = {
        "role": "assistant",
        "content": conteudo_final,
        "model": modelo,
        "skill": skill_id,
        "route": rota.rota,
        "confidence": rota.confianca,
        "reason": rota.motivo,
        "router_metadata": rota.metadata,
        "plan": serializar_plano(plano),
        "timestamp": datetime.now().isoformat(),
    }

    mensagens_chat.append(
        mensagem_assistente_final,
    )

    chat["messages"] = mensagens_chat

    chat["updated_at"] = (
        datetime.now().isoformat()
    )

    storage.salvar_chat(
        chat,
    )

    # --------------------------------------------------------
    # SSE FINAL
    # --------------------------------------------------------

    yield (
        "data: "
        + json.dumps(
            {
                "type": "message",
                "content": conteudo_final,
                "model": modelo,
                "route": rota.rota,
                "skill": skill_id,
                "confidence": rota.confianca,
                "reason": rota.motivo,
                "router_metadata": rota.metadata,
                "plan": serializar_plano(plano),
            },
            ensure_ascii=False,
        )
        + "\n\n"
    )

    yield (
        "data: "
        + json.dumps(
            {
                "type": "done",
                "chat_id": chat_id,
            },
            ensure_ascii=False,
        )
        + "\n\n"
    )

from dataclasses import dataclass, field
from typing import Any


@dataclass
class ModelContract:
    """
    Representa um modelo de IA de forma independente
    do provider utilizado para executÃ¡-lo.
    """

    id: str
    nome: str
    provider: str = "ollama"

    descricao: str = ""

    suporta_tools: bool = False
    suporta_vision: bool = False

    contexto_maximo: int | None = None

    capacidades: list[str] = field(
        default_factory=list
    )

    metadata: dict[str, Any] = field(
        default_factory=dict
    )

    def validar(self) -> tuple[bool, str]:
        if not self.id.strip():
            return False, (
                "O modelo precisa possuir um ID."
            )

        if not self.nome.strip():
            return False, (
                "O modelo precisa possuir um nome."
            )

        if not self.provider.strip():
            return False, (
                "O modelo precisa possuir um provider."
            )

        if self.contexto_maximo is not None:
            if self.contexto_maximo <= 0:
                return False, (
                    "O contexto mÃ¡ximo deve ser maior que zero."
                )

        if not isinstance(self.capacidades, list):
            return False, (
                "As capacidades devem ser uma lista."
            )

        if not isinstance(self.metadata, dict):
            return False, (
                "Os metadados devem ser um objeto."
            )

        return True, ""

class ModelProvider(ABC):
    """
    Interface comum para execução de modelos de IA.
    """

    @abstractmethod
    def gerar(
        self,
        modelo: str,
        mensagens: list[dict],
        temperatura: float = 0.5,
        ferramentas: list[dict] | None = None,
    ) -> dict:
        raise NotImplementedError


class OllamaModelProvider(ModelProvider):
    """
    Provider HTTP para Ollama.
    """

    def __init__(
        self,
        url: str | None = None,
        timeout: int = 300,
    ):
        self.url = url or os.getenv(
            "OLLAMA_URL",
            "http://localhost:11434/api/chat",
        )
        self.timeout = timeout

    def gerar(
        self,
        modelo: str,
        mensagens: list[dict],
        temperatura: float = 0.5,
        ferramentas: list[dict] | None = None,
    ) -> dict:
        payload: dict[str, Any] = {
            "model": modelo,
            "messages": mensagens,
            "stream": False,
            "options": {
                "temperature": temperatura,
            },
        }

        if ferramentas:
            payload["tools"] = ferramentas

        resposta = requests.post(
            self.url,
            json=payload,
            timeout=self.timeout,
        )

        resposta.raise_for_status()

        dados = resposta.json()

        if not isinstance(dados, dict):
            raise RuntimeError(
                "O provedor Ollama retornou uma resposta inválida."
            )

        return dados


def criar_model_provider() -> ModelProvider:
    provider = os.getenv(
        "MODEL_PROVIDER",
        "ollama",
    ).strip().lower()

    if provider == "ollama":
        return OllamaModelProvider()

    raise ValueError(
        f"Model Provider não suportado: '{provider}'."
    )

class ModelRegistry:
    """
    Registro central dos modelos disponíveis no Agent Core.
    """

    def __init__(self):
        self._modelos: dict[str, ModelContract] = {}

    def registrar(
        self,
        modelo: ModelContract,
    ) -> None:
        valido, erro = modelo.validar()

        if not valido:
            raise ValueError(
                f"Modelo inválido: {erro}"
            )

        if modelo.id in self._modelos:
            raise ValueError(
                f"O modelo '{modelo.id}' já está registrado."
            )

        self._modelos[modelo.id] = modelo

    def obter(
        self,
        modelo_id: str,
    ) -> ModelContract | None:
        return self._modelos.get(modelo_id)

    def selecionar(
        self,
        modelo_id: str,
        fallback_id: str = "llama3.2:1b",
    ) -> tuple[ModelContract, bool]:
        """
        Seleciona um modelo registrado.

        Retorna:
            (modelo, fallback_utilizado)
        """

        modelo = self.obter(modelo_id)

        if modelo is not None:
            return modelo, False

        fallback = self.obter(fallback_id)

        if fallback is None:
            raise ValueError(
                f"O modelo '{modelo_id}' não está registrado "
                f"e o fallback '{fallback_id}' também não está disponível."
            )

        return fallback, True

    def existe(
        self,
        modelo_id: str,
    ) -> bool:
        return modelo_id in self._modelos

    def listar(self) -> list[ModelContract]:
        return list(self._modelos.values())

    def remover(
        self,
        modelo_id: str,
    ) -> bool:
        if modelo_id not in self._modelos:
            return False

        del self._modelos[modelo_id]
        return True

    def limpar(self) -> None:
        self._modelos.clear()

class Database:
    def __init__(self, path: Path):
        self.path = path

    def initialize(self) -> None:
        self.path.parent.mkdir(parents=True, exist_ok=True, mode=0o700)
        with sqlite3.connect(self.path, timeout=5) as connection:
            version = connection.execute("PRAGMA user_version").fetchone()[0]
            if version not in (0, 1):
                raise RuntimeError("VersÃ£o de banco incompatÃ­vel")
            connection.execute("PRAGMA journal_mode=WAL")
            connection.execute("CREATE TABLE IF NOT EXISTS app_metadata (key TEXT PRIMARY KEY, value TEXT NOT NULL)")
            connection.execute("INSERT OR IGNORE INTO app_metadata VALUES ('schema_version', '1')")
            connection.execute("PRAGMA user_version=1")
        self.path.chmod(0o600)

    def ready(self) -> bool:
        try:
            uri = self.path.resolve().as_uri() + "?mode=rw"
            with sqlite3.connect(uri, uri=True, timeout=2) as connection:
                return connection.execute("SELECT value FROM app_metadata WHERE key='schema_version'").fetchone() == ('1',)
        except sqlite3.Error:
            return False

        )

                "extensao": caminho.suffix.lower(),
                "nome_arquivo": caminho.name,
                "caminho": str(caminho),
            },
        )

        valido, erro = documento.validar()

        if not valido:
            raise ValueError(

            )

class EmbeddingProvider(ABC):

    @abstractmethod
    def gerar(
        self,
        texto: str,
    ) -> list[float]:
        raise NotImplementedError



class HashEmbeddingProvider(
    EmbeddingProvider
):
    def __init__(
        self,
        dimensao: int = 32,
    ):
        self.dimensao = dimensao

    def gerar(
        self,
        texto: str,
    ) -> list[float]:
        import hashlib

        valores = []

        for indice in range(
            self.dimensao
        ):
            digest = hashlib.sha256(
                f"{indice}:{texto}".encode(
                    "utf-8"
                )
            ).digest()

            valor = int.from_bytes(
                digest[:4],
                byteorder="big",
            )

            valores.append(
                valor / 4_294_967_295
            )

        return valores

class MemoryRetriever:
    def __init__(self, store: MemoryStore):
        self.store = store

    def buscar(
        self,
        consulta: str,
        limite: int = 5,
    ) -> list[MemoryContract]:
        consulta_normalizada = self._normalizar(
            consulta
        )

        if not consulta_normalizada:
            return []

        if limite <= 0:
            return []

        termos = set(
            consulta_normalizada.split()
        )

        memorias = self.store.listar()
        resultados = []

        for memoria in memorias:
            texto = self._normalizar(
                memoria.conteudo
            )

            palavras = set(
                texto.split()
            )

            correspondencias = len(
                termos & palavras
            )

            if correspondencias == 0:
                continue

            pontuacao = (
                correspondencias
                + memoria.importancia
            )

            resultados.append(
                (
                    pontuacao,
                    memoria,
                )
            )

        resultados.sort(
            key=lambda item: item[0],
            reverse=True,
        )

        return [
            memoria
            for _, memoria in resultados[:limite]
        ]

    def buscar_por_tipo(
        self,
        tipo: str,
        limite: int = 5,
    ) -> list[MemoryContract]:
        if limite <= 0:
            return []

        memorias = [
            memoria
            for memoria in self.store.listar()
            if memoria.tipo == tipo
        ]

        memorias.sort(
            key=lambda memoria: memoria.importancia,
            reverse=True,
        )

        return memorias[:limite]

    def _normalizar(
        self,
        texto: str,
    ) -> str:
        texto = texto.lower()

        texto = re.sub(
            r"[^\w\s]",
            " ",
            texto,
            flags=re.UNICODE,
        )

        texto = re.sub(
            r"\s+",
            " ",
            texto,
        )

        return texto.strip()

class RAGRetriever:
    def __init__(
        self,
        vector_store: VectorStore,
        embedding_provider: EmbeddingProvider,
    ):
        self.vector_store = vector_store
        self.embedding_provider = embedding_provider

    def buscar(
        self,
        consulta: str,
        limite: int = 5,
    ) -> list[VectorDocument]:
        consulta = consulta.strip()

        if not consulta:
            return []

        if limite <= 0:
            return []

        embedding_consulta = (
            self.embedding_provider.gerar(
                consulta
            )
        )

        documentos = self.vector_store.listar()

        resultados = []

        for documento in documentos:
            similaridade = self._similaridade_cosseno(
                embedding_consulta,
                documento.embedding,
            )

            resultados.append(
                (
                    similaridade,
                    documento,
                )
            )

        resultados.sort(
            key=lambda item: item[0],
            reverse=True,
        )

        return [
            documento
            for _, documento in resultados[:limite]
        ]

    def buscar_com_pontuacao(
        self,
        consulta: str,
        limite: int = 5,
    ) -> list[tuple[VectorDocument, float]]:
        consulta = consulta.strip()

        if not consulta:
            return []

        if limite <= 0:
            return []

        embedding_consulta = (
            self.embedding_provider.gerar(
                consulta
            )
        )

        resultados = []

        for documento in self.vector_store.listar():
            similaridade = self._similaridade_cosseno(
                embedding_consulta,
                documento.embedding,
            )

            resultados.append(
                (
                    documento,
                    similaridade,
                )
            )

        resultados.sort(
            key=lambda item: item[1],
            reverse=True,
        )

        return resultados[:limite]

    def _similaridade_cosseno(
        self,
        vetor_a: list[float],
        vetor_b: list[float],
    ) -> float:
        if len(vetor_a) != len(vetor_b):
            raise ValueError(
                "Os embeddings precisam possuir "
                "a mesma dimensÃ£o."
            )

        produto_escalar = sum(
            a * b
            for a, b in zip(
                vetor_a,
                vetor_b,
            )
        )

        norma_a = math.sqrt(
            sum(
                valor * valor
                for valor in vetor_a
            )
        )

        norma_b = math.sqrt(
            sum(
                valor * valor
                for valor in vetor_b
            )
        )

        if norma_a == 0 or norma_b == 0:
            return 0.0

        return (
            produto_escalar
            / (norma_a * norma_b)
        )

class OllamaProvider:
    def __init__(self, settings: Settings, transport: httpx.AsyncBaseTransport | None = None):
        self.settings = settings
        self.client = httpx.AsyncClient(base_url=settings.ollama_url, trust_env=False,
                                       follow_redirects=False, transport=transport)

    async def health(self) -> ModelHealth:
        status = "ready"
        try:
            response = await self.client.get("/api/tags", timeout=self.settings.health_timeout)
            response.raise_for_status()
            models = response.json()["models"]
            if not isinstance(models, list) or any(not isinstance(m, dict) or not isinstance(m.get("name"), str) for m in models):
                raise ValueError("invalid models")
            if self.settings.model not in {m["name"] for m in models}:
                status = "model_missing"
        except httpx.HTTPError:
            status = "unreachable"
        except (ValueError, KeyError, TypeError):
            status = "invalid_response"
        return ModelHealth(status=status, model=self.settings.model)

    async def chat(self, messages: list[Message]) -> ChatResult:
        if not messages:
            raise ProviderError("empty_messages")
        payload = {"model": self.settings.model, "messages": [m.model_dump() for m in messages],
                   "stream": False, "keep_alive": "1m",
                   "options": {"num_ctx": self.settings.context_window,
                               "num_predict": self.settings.max_output_tokens}}
        try:
            response = await self.client.post("/api/chat", json=payload,
                                              timeout=self.settings.generation_timeout)
            response.raise_for_status()
            body = response.json()
            content = body["message"]["content"]
            if body.get("done") is not True or not isinstance(content, str) or not content.strip():
                raise ValueError("incomplete response")
            return ChatResult(content=content, model=self.settings.model)
        except httpx.HTTPError:
            raise ProviderError("ollama_request_failed") from None
        except (KeyError, TypeError, ValueError):
            raise ProviderError("ollama_invalid_response") from None

    async def aclose(self) -> None:
        await self.client.aclose()

            if permissao.contexto_id == contexto_id
        ]

    def remover(
        self,
        contexto_id: str,
        ferramenta_id: str,
    ) -> bool:
        chave = self._gerar_chave(
            contexto_id,
            ferramenta_id,
        )

        if chave not in self._permissoes:
            return False

        del self._permissoes[chave]
        return True

    def limpar(self) -> None:
        self._permissoes.clear()

    def _gerar_chave(
        self,
        contexto_id: str,
        ferramenta_id: str,
    ) -> str:
        return f"{contexto_id}:{ferramenta_id}"

            return permissao

        permissao = self.engine.avaliar(

            ferramenta,
        )

        return permissao

            ferramenta_id,

    ) -> None:

        )

            self.permission_registry.remover(
                permissao.contexto_id,
                permissao.ferramenta_id,
            )

        self.permission_registry.registrar(
            permissao
        )

    objetivo: str
    etapas: list[PlanStep] = field(default_factory=list)
    metadata: dict[str, Any] = field(default_factory=dict)

    def validar(self) -> tuple[bool, str]:
        if not self.objetivo.strip():
            return False, "O plano precisa possuir um objetivo."

        ids: set[str] = set()

        for etapa in self.etapas:
            if not etapa.id:
                return False, "Toda etapa precisa possuir um ID."

            if etapa.id in ids:
                return False, (
                    f"A etapa '{etapa.id}' estÃ¡ duplicada."
                )

            ids.add(etapa.id)

            if not etapa.descricao.strip():
                return False, (
                    f"A etapa '{etapa.id}' precisa possuir uma descriÃ§Ã£o."
                )

            if etapa.estado not in ESTADOS_ETAPA_VALIDOS:
                return False, (
                    f"A etapa '{etapa.id}' possui estado invÃ¡lido: "
                    f"'{etapa.estado}'."
                )

        return True, ""

class Router:
    """
    Roteador central do Agent Core.

    O Router decide qual fluxo deve processar
    uma solicitação, mas não executa o modelo.

    A seleção do modelo é feita através do
    ModelRegistry.
    """

    def __init__(
        self,
        model_registry: ModelRegistry | None = None,
        modelo_simples: str = MODELO_FALLBACK,
        modelo_codigo: str = MODELO_FALLBACK,
        modelo_complexo: str = MODELO_FALLBACK,
    ):
        self.model_registry = (
            model_registry
            or criar_registry_padrao()
        )

        self.modelo_simples = modelo_simples
        self.modelo_codigo = modelo_codigo
        self.modelo_complexo = modelo_complexo

    # ==========================================================
    # API PRINCIPAL
    # ==========================================================

    def rotear(
        self,
        texto: str,
        skill_id: str | None = None,
    ) -> RouterContract:

        texto_normalizado = self._normalizar(
            texto
        )

        if not texto_normalizado:
            return self._criar_resultado(
                rota="simples",
                skill_id=skill_id,
                motivo="Mensagem vazia ou sem conteúdo.",
                confianca=1.0,
            )

        rota, confianca, motivo = (
            self._determinar_rota(
                texto_normalizado
            )
        )

        return self._criar_resultado(
            rota=rota,
            skill_id=skill_id,
            motivo=motivo,
            confianca=confianca,
        )

    # ==========================================================
    # NORMALIZAÇÃO
    # ==========================================================

    def _normalizar(
        self,
        texto: str,
    ) -> str:

        if not isinstance(texto, str):
            return ""

        return texto.strip().lower()

    # ==========================================================
    # DETERMINAÇÃO DA ROTA
    # ==========================================================

    def _determinar_rota(
        self,
        texto: str,
    ) -> tuple[str, float, str]:

        if self._parece_codigo(texto):
            return (
                "codigo",
                0.9,
                "A solicitação aparenta envolver programação.",
            )

        if self._parece_complexo(texto):
            return (
                "complexo",
                0.7,
                "A solicitação aparenta exigir processamento mais complexo.",
            )

        return (
            "simples",
            0.8,
            "Solicitação geral sem indicação de complexidade elevada.",
        )

    # ==========================================================
    # RESULTADO
    # ==========================================================

    def _criar_resultado(
        self,
        rota: str,
        skill_id: str | None,
        motivo: str,
        confianca: float,
    ) -> RouterContract:

        if rota not in ROTAS_VALIDAS:
            raise ValueError(
                f"Rota inválida: '{rota}'."
            )

        modelo, fallback_utilizado = (
            self._modelo_da_rota(rota)
        )

        metadata = {
            "router_version": "3.0",
            "deterministico": True,
            "model_registry": True,
            "fallback_utilizado": fallback_utilizado,
        }

        return RouterContract(
            rota=rota,
            modelo=modelo.id,
            skill_id=skill_id,
            motivo=motivo,
            confianca=confianca,
            metadata=metadata,
        )

    # ==========================================================
    # MODELO DA ROTA
    # ==========================================================

    def _modelo_da_rota(
        self,
        rota: str,
    ):
        modelos = {
            "simples": self.modelo_simples,
            "codigo": self.modelo_codigo,
            "complexo": self.modelo_complexo,
        }

        try:
            modelo_id = modelos[rota]
        except KeyError:
            raise ValueError(
                f"Não existe modelo configurado "
                f"para a rota '{rota}'."
            )

        return self.model_registry.selecionar(
            modelo_id=modelo_id,
            fallback_id=MODELO_FALLBACK,
        )

    # ==========================================================
    # DETECÇÃO DE CÓDIGO
    # ==========================================================

    def _parece_codigo(
        self,
        texto: str,
    ) -> bool:

        indicadores = (
            "python",
            "java",
            "javascript",
            "typescript",
            "kotlin",
            "php",
            "c++",
            "c#",
            "html",
            "css",
            "sql",
            "debug",
            "debugar",
            "erro no código",
            "erro no codigo",
            "stack trace",
            "exception",
            "traceback",
            "função",
            "funcao",
            "classe",
            "método",
            "metodo",
            "api",
            "endpoint",
            "fastapi",
            "android",
            "gradle",
            "docker",
            "git",
        )

        return any(
            indicador in texto
            for indicador in indicadores
        )

    # ==========================================================
    # DETECÇÃO DE COMPLEXIDADE
    # ==========================================================

    def _parece_complexo(

    ) -> bool:

        indicadores = (

def ler_frontmatter(conteudo: str) -> tuple[dict[str, Any], str]:
    """
    Extrai o Frontmatter YAML de uma Skill Markdown.

    Retorna:
        (configuracoes, corpo)
    """

    conteudo = conteudo.strip()

    if not conteudo.startswith("---"):
        return {}, conteudo

    partes = conteudo.split("---", 2)

    if len(partes) != 3:
        return {}, conteudo

    try:
        configuracoes = yaml.safe_load(partes[1]) or {}

        if not isinstance(configuracoes, dict):
            configuracoes = {}

        corpo = partes[2].strip()

        return configuracoes, corpo

    except yaml.YAMLError as erro:
        print(
            f"Erro ao interpretar Frontmatter YAML: {erro}"
        )
        return {}, conteudo


def normalizar_ferramentas(valor: Any) -> list[str]:
    """
    Garante que ferramentas_permitidas seja uma lista
    limpa de strings Ãºnicas.
    """

    if valor is None:
        return []

    if isinstance(valor, str):
        valor = valor.strip()

        return [valor] if valor else []

    if not isinstance(valor, list):
        return []

    resultado = []

    for item in valor:
        if not isinstance(item, str):
            continue

        item = item.strip()

        if item and item not in resultado:
            resultado.append(item)

    return resultado


def criar_configuracao(

) -> SkillFrontmatter:

    ferramentas = normalizar_ferramentas(

    temperatura = configuracoes.get(

    )

    try:
        temperatura = float(temperatura)
    except (TypeError, ValueError):
        temperatura = 0.5

    return SkillFrontmatter(
        id=(
            str(configuracoes["id"]).strip()
            if configuracoes.get("id") is not None

        ),
        nome=(
            str(configuracoes["nome"]).strip()
            if configuracoes.get("nome") is not None

            if configuracoes.get("modelo") is not None

IGNORAR_DIRETORIOS = {
    "assets",
    "images",
    "docs",
    "__pycache__",
}


class SkillLoader:
    def __init__(self, skills_dir: Path):
        self.skills_dir = skills_dir

    def descobrir_arquivos(self) -> list[Path]:
        if not self.skills_dir.exists():
            return []

        arquivos = []

        for caminho in self.skills_dir.rglob("*"):
            if not caminho.is_file():
                continue

            if caminho.suffix.lower() not in {".md", ".txt"}:
                continue

            if any(
                parte.lower() in IGNORAR_DIRETORIOS
                for parte in caminho.relative_to(self.skills_dir).parts[:-1]
            ):
                continue

            arquivos.append(caminho)

        return sorted(
            arquivos,
            key=lambda caminho: str(caminho).casefold()
        )

    def identificar_id(self, caminho: Path) -> str:
        # Futuramente:
        # skills/minha_skill/SKILL.md -> minha_skill
        if caminho.name.lower() == "skill.md":
            return caminho.parent.name

        # Compatibilidade atual:
        # skills/minha_skill.md -> minha_skill
        return caminho.stem

    def carregar_arquivo(self, caminho: Path) -> SkillArquivo:
        conteudo = caminho.read_text(
            encoding="utf-8"
        )

        return SkillArquivo(
            id=self.identificar_id(caminho),
            caminho=caminho,
            conteudo=conteudo,
        )

    def carregar_todas(self) -> list[SkillArquivo]:
        skills = []

        for caminho in self.descobrir_arquivos():
            try:
                skills.append(
                    self.carregar_arquivo(caminho)
                )
            except (OSError, UnicodeError):
                continue

        return skills

class ExecutionGuard:

    """

class FerramentaTerminal(FerramentaBase):

    def executar(
        self,
        argumentos: dict
    ) -> str:

        comando = argumentos.get(
            "comando",
            ""
        )

        try:

            resultado = subprocess.run(
                comando,
                shell=True,
                capture_output=True,
                text=True,
                timeout=30
            )

            saida = resultado.stdout.strip()
            erro = resultado.stderr.strip()

            if erro and not saida:

                return (
                    "Comando executado com "
                    "cÃ³digo de erro:\n"
                    f"{erro}"
                )

            elif erro and saida:

                return (
                    f"SaÃ­da:\n{saida}\n\n"
                    "Erros/Avisos gerados:\n"
                    f"{erro}"
                )

            elif saida:

                if len(saida) > 4000:

                    return (
                        "SaÃ­da do terminal "
                        "(truncada por seguranÃ§a):\n"
                        f"{saida[:4000]}..."
                    )

                return (
                    f"SaÃ­da do terminal:\n"
                    f"{saida}"
                )

            else:

                return (
                    f"O comando '{comando}' "
                    "foi executado com sucesso, "
                    "mas nÃ£o retornou texto."
                )

        except subprocess.TimeoutExpired:

            return (
                "Erro: O comando demorou mais "
                "de 30 segundos e foi interrompido."
            )

        except Exception as erro:

            return (
                "Erro fatal ao tentar usar "
                f"o terminal: {erro}"
            )

class ToolPermissionPolicy:
    def __init__(self):
        self._permissoes: dict[str, set[str]] = {}

    def definir_permissoes(
        self,
        contexto_id: str,
        ferramentas: list[str],
    ) -> None:
        self._permissoes[contexto_id] = set(ferramentas)

    def permitir(
        self,
        contexto_id: str,
        ferramenta: ToolContract,
    ) -> bool:
        ferramentas_permitidas = self._permissoes.get(
            contexto_id,
            set(),
        )

        return ferramenta.id in ferramentas_permitidas

    def listar_permitidas(
        self,
        contexto_id: str,
    ) -> list[str]:
        return sorted(
            self._permissoes.get(contexto_id, set())
        )

    def remover_contexto(self, contexto_id: str) -> None:
        self._permissoes.pop(contexto_id, None)

    def limpar(self) -> None:
        self._permissoes.clear()

    def carregar_skill(
        self,
        contexto_id: str,
        ferramentas_permitidas: list[str] | None,
    ) -> None:
        self.definir_permissoes(
            contexto_id,
            ferramentas_permitidas or [],
        )

class ToolRegistry:
    """
    Registro central de todas as ferramentas disponÃ­veis
    no Agent Core.

    Responsabilidades:
    - registrar ferramentas
    - localizar ferramentas
    - listar ferramentas
    - remover ferramentas
    - gerar schemas para o Ollama
    """

    def __init__(self):
        self._ferramentas: dict[str, ToolContract] = {}

    # ==========================================================
    # REGISTRO
    # ==========================================================

    def registrar(
        self,
        ferramenta: ToolContract
    ) -> None:

        if not ferramenta.id:
            raise ValueError(
                "A ferramenta precisa possuir um ID."
            )

        if ferramenta.id in self._ferramentas:
            raise ValueError(
                f"A ferramenta '{ferramenta.id}' "
                "jÃ¡ estÃ¡ registrada."
            )

        self._ferramentas[ferramenta.id] = ferramenta

    # ==========================================================
    # CONSULTA
    # ==========================================================

    def obter(
        self,
        ferramenta_id: str
    ) -> ToolContract | None:

        return self._ferramentas.get(
            ferramenta_id
        )

    def existe(
        self,
        ferramenta_id: str
    ) -> bool:

        return ferramenta_id in self._ferramentas

    def listar(self) -> list[ToolContract]:

        return list(
            self._ferramentas.values()
        )

    # ==========================================================
    # REMOÃ‡ÃƒO
    # ==========================================================

    def remover(
        self,
        ferramenta_id: str
    ) -> bool:

        if ferramenta_id not in self._ferramentas:
            return False

        del self._ferramentas[
            ferramenta_id
        ]

        return True

    def limpar(self) -> None:

        self._ferramentas.clear()

    # ==========================================================
    # SCHEMA OLLAMA
    # ==========================================================

    def gerar_schema_ollama(
        self,
        ferramenta: ToolContract
    ) -> dict:

        propriedades = {}

        obrigatorios = []

        for nome, configuracao in (
            ferramenta.argumentos or {}
        ).items():

            if not isinstance(
                configuracao,
                dict
            ):
                continue

            propriedade = {
                "type": configuracao.get(
                    "tipo",
                    "string"
                )
            }

            descricao = configuracao.get(
                "descricao"
            )

            if descricao:
                propriedade[
                    "description"
                ] = descricao

            enum = configuracao.get(
                "enum"
            )

            if enum:
                propriedade[
                    "enum"
                ] = enum

            propriedades[
                nome
            ] = propriedade

            if configuracao.get(
                "obrigatorio",
                False
            ):
                obrigatorios.append(
                    nome
                )

        parameters = {
            "type": "object",
            "properties": propriedades,
        }

        if obrigatorios:
            parameters[
                "required"
            ] = obrigatorios

        return {
            "type": "function",
            "function": {
                "name": ferramenta.id,
                "description": ferramenta.descricao,
                "parameters": parameters,
            },
        }

    # ==========================================================
    # SCHEMAS DE VÃRIAS FERRAMENTAS
    # ==========================================================

    def gerar_schemas_ollama(
        self,
        ferramenta_ids: list[str] | None = None
    ) -> list[dict]:
        """
        Gera os schemas Ollama das ferramentas solicitadas.

        Se nenhuma lista for informada, gera o schema
        de todas as ferramentas registradas.
        """

        if ferramenta_ids is None:

            ferramentas = self.listar()

        else:

            ferramentas = []

            for ferramenta_id in ferramenta_ids:

                ferramenta = self.obter(
                    ferramenta_id
                )

                if ferramenta is not None:
                    ferramentas.append(
                        ferramenta
                    )

        return [
            self.gerar_schema_ollama(
                ferramenta
            )
            for ferramenta in ferramentas
        ]

TIPOS_PERMITIDOS = {
    "string": str,
    "integer": int,
    "number": (int, float),
    "boolean": bool,
    "object": dict,
    "array": list,
}


def validar_argumentos(
    ferramenta: ToolContract,
    argumentos: dict[str, Any],
) -> tuple[bool, str]:
    if not isinstance(argumentos, dict):
        return False, "Os argumentos da ferramenta devem ser um objeto."

    schema = ferramenta.argumentos or {}

    for nome, configuracao in schema.items():
        if not isinstance(configuracao, dict):
            continue

        obrigatorio = configuracao.get("obrigatorio", False)

        if obrigatorio and nome not in argumentos:
            return False, (
                f"O argumento obrigatÃ³rio '{nome}' nÃ£o foi informado."
            )

        if nome not in argumentos:
            continue

        valor = argumentos[nome]
        tipo = configuracao.get("tipo")

        if tipo not in TIPOS_PERMITIDOS:
            return False, (
                f"O schema da ferramenta possui o tipo "
                f"desconhecido '{tipo}'."
            )

        tipo_esperado = TIPOS_PERMITIDOS[tipo]

        # bool Ã© uma subclasse de int em Python.
        # Por isso precisa ser tratado separadamente.
        if tipo in {"integer", "number"} and isinstance(valor, bool):
            return False, (
                f"O argumento '{nome}' deve possuir o tipo '{tipo}'."
            )

        if not isinstance(valor, tipo_esperado):
            return False, (
                f"O argumento '{nome}' deve possuir o tipo '{tipo}'."
            )

    return True, ""
